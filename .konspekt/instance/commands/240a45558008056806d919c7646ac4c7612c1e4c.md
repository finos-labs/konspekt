import re,html
s=open('docs/index.html').read()
secs=re.split(r'<section',s)[1:]
print(len(secs))
for i,sec in enumerate(secs,1):
    t=re.sub(r'<script.*?</script>|<style.*?</style>','',sec,flags=re.S)
    hs=re.findall(r'<h[1-3][^>]*>(.*?)</h[1-3]>',t,re.S)
    txt=html.unescape(re.sub(r'<[^>]+>',' ',t)); txt=re.sub(r'\s+',' ',txt)
    print(f"--- {i}: {' | '.join(html.unescape(re.sub('<[^>]+>','',h)).strip() for h in hs)[:150]}")
    print("   ",txt[:420])
EOF
