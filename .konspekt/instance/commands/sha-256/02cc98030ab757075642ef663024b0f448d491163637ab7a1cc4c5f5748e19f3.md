cd /c/Users/denis/IdeaProjects/konspekt && python - <<'EOF'
p='.gitattributes'
s=open(p,encoding='utf-8').read()
a=s[s.index('# setup/init.mjs copies these templates'):]
b='''# setup/ is the adopter kit. setup/init.mjs copies its templates and component
# files byte for byte into an adopter's .konspekt/instance/, .konspekt/ and
# .github/workflows/, and distribution/latest/ is rebuilt from it. Pin the whole
# directory to LF so a scaffold or a distribution build run from a Windows
# checkout writes LF files, matching the instance rule above, and so the
# policy-field check in test/setup-init.test.mjs reads the seeded project.md the
# same way on every platform.
/setup/** text eol=lf
'''
s=s.replace(a,b)
open(p,'w',encoding='utf-8',newline='').write(s)
EOF
for f in $(git ls-files --eol setup | grep "w/crlf" | awk '{print $NF}'); do rm -f "$f"; done; git checkout -- setup && git ls-files --eol setup | awk '{print $2}' | sort | uniq -c; git status --short; node test/setup-init.test.mjs 2>&1 | tail -1
